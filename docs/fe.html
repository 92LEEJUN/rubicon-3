// 커밋 결과로 게이트 상태 토글 function apply(res: CommitResult, kind) { if (res.status === 'confirm') { // 409 pending.current = { kind, payload: res.payload }; setConfirmTemplate(res.template); track('checkout_shown', { kind }); } else if (res.status === 'login') { // 401 setShowLogin(true); } else if (res.status === 'ok') { track('order_confirmed', { kind }); opts?.onCommitted?.(kind, res.data); } } // start(cta)→confirmed:false · confirm()→confirmed:true 재제출 · login()→토큰 후 재시도
